import { components, STATUS_CONFIG } from '../data/component-inventory';

export const base = import.meta.env.BASE_URL.replace(/\/$/, '');

export const tableHeader = {
  col1: 'Component',
  col2: 'Platforms',
  col3: 'Web',
  col4: 'Flutter',
};

export const tableRows = components.map((c) => {
  const webCfg = STATUS_CONFIG[c.web];
  const flutterCfg = STATUS_CONFIG[c.flutter];
  const hasWeb = c.web !== 'not-available';
  const hasFlutter = c.flutter !== 'not-available';

  const nameCell = c.href
    ? `<a href="${base}${c.href}" style="color: var(--md-sys-color-primary); text-decoration: none; font-weight: 500;">${c.name}</a>`
    : c.name;

  const platformCell = [
    hasWeb ? `<asm-icon name="language" style="font-size: 18px; color: var(--md-sys-color-primary);"></asm-icon>` : '',
    hasFlutter ? `<asm-icon name="flutter" style="font-size: 18px; color: var(--md-sys-color-secondary);"></asm-icon>` : '',
  ].filter(Boolean).join(' ');

  const webCell = `<asm-badge label="${webCfg.label}" status="${webCfg.status}"></asm-badge>`;
  const flutterCell = `<asm-badge label="${flutterCfg.label}" status="${flutterCfg.status}"></asm-badge>`;

  return { col1: nameCell, col2: platformCell, col3: webCell, col4: flutterCell };
});

  (() => {
    const table = document.getElementById('components-table');
    if (table) {
      customElements.whenDefined('asm-table').then(() => {
        table.header = ${JSON.stringify(tableHeader)};
        table.rows = ${JSON.stringify(tableRows)};
      });

  })();
`}</script>
